4 — Le serveur Borne

borne/routes/videos.js — upload

Le cœur de la Borne : recevoir les vidéos enregistrées par les invités, les servir, les télécharger, les supprimer. Cette page concentre trois invariants critiques : le fileFilter tolérant pour Safari (§11.4), la transaction de remplacement avec unlink après commit (§11.9), et le streaming Range-aware (§11.3).

multer + le fileFilter Safari

const VIDEO_EXTENSIONS = new Set(['.mp4', '.webm', '.mov', '.avi', '.mkv']);

function makeMulter(dataDir) {
  const storage = multer.diskStorage({
    destination(req, file, cb) {
      const active = getActiveEvent();
      if (!active) return cb(new Error('Aucun événement actif'));
      cb(null, join(dataDir, 'events', active.id, 'videos'));
    },
    filename(req, file, cb) {
      const ext = extname(file.originalname).toLowerCase() || '.mp4';
      cb(null, `${uuidv4()}${ext}`);                  // nom unique, on ne fait pas confiance au nom client
    },
  });

  // fileFilter : tolère video/* ET les mimes génériques Safari avec extension vidéo (§11.4)
  const fileFilter = (req, file, cb) => {
    const mime = file.mimetype;
    const ext = extname(file.originalname).toLowerCase();
    const isVideoMime = mime.startsWith('video/');
    const isGenericMime = mime === 'text/plain' || mime === 'application/octet-stream';
    if (isVideoMime || (isGenericMime && VIDEO_EXTENSIONS.has(ext))) cb(null, true);
    else cb(Object.assign(new Error('Type de fichier non supporté'), { status: 400 }));
  };

  return multer({ storage, fileFilter, limits: { fileSize: LIMITS.VIDEO_MAX_BYTES } });
}

⛔ Invariant §11.4 : le fileFilter doit tolérer les MIME génériques de Safari

Quand l'iPad (Safari) uploade une vidéo enregistrée via MediaRecorder, il n'envoie pas toujours un type MIME video/* : selon les versions, il met application/octet-stream ou même text/plain. Un filtre naïf qui n'accepterait que video/* rejetterait les uploads iPad — or l'iPad est la cible principale. La parade : accepter aussi les MIME génériques à condition que l'extension du fichier soit une extension vidéo connue (VIDEO_EXTENSIONS). On combine donc deux signaux faibles (MIME + extension) pour une décision robuste. C'est un invariant Safari spécifique, facile à casser si on « durcit » le filtre sans connaître ce contexte.

Le filename génère un nom unique (uuidv4() + ext) : on ne réutilise jamais le nom du fichier client (collision, caractères dangereux). L'extension est forcée en minuscules avec repli .mp4.

L'upload : ordre des gardes et nettoyage du fichier

router.post('/videos', upload.single('video'), async (req, res, next) => {
  // multer a déjà stocké le fichier sur disque si on arrive ici
  const ctx = requireActiveDb(res);
  if (!ctx) {
    if (req.file) unlink(req.file.path, () => {});   // nettoie le fichier orphelin
    return;
  }
  const { active, db } = ctx;
  const { session_id, question_id, question_text, recorded_at } = req.body;

  if (!req.file) return res.status(400).json({ error: 'Fichier vidéo requis' });
  if (!session_id)   { unlink(req.file.path, () => {}); return res.status(400).json({ error: 'session_id est requis' }); }
  if (!question_text){ unlink(req.file.path, () => {}); return res.status(400).json({ error: 'question_text est requis' }); }

  const session = db.prepare('SELECT id FROM sessions WHERE id=?').get(session_id);
  if (!session) { unlink(req.file.path, () => {}); return res.status(400).json({ error: 'Session introuvable' }); }
  // … suite : checksum + transaction (voir plus bas) …
});
⚠ Pourquoi unlink à chaque garde ?

multer écrit le fichier avant que le handler ne s'exécute (le middleware upload.single('video') tourne en premier). Donc si une validation échoue ensuite (pas de session_id, session inexistante…), le fichier est déjà sur le disque — il faut le supprimer pour ne pas accumuler des orphelins. D'où l'unlink répété. L'ordre des gardes compte : on teste req.file en premier car les suivantes accèdent à req.file.path.

JS fs.unlink en style callback

unlink(path, () => {}) supprime un fichier. Ici on utilise la version callback de fs (pas await) avec un callback vide () => {} : « supprime, et peu importe le résultat ». C'est un nettoyage best-effort, fire-and-forget — on ne veut pas que l'échec d'un unlink (fichier déjà parti) interrompe la réponse d'erreur en cours. À comparer avec unlinkSync (synchrone) utilisé ailleurs quand on veut attendre.

La transaction de remplacement (§11.9)

const checksum = await sha256File(req.file.path);
const id = uuidv4();
const filename = req.file.filename;
const mime_type = req.file.mimetype.startsWith('video/') ? req.file.mimetype : 'video/mp4';
const size = req.file.size;

// Remplacement transactionnel §5.2 + §11.9 : DELETE+INSERT en transaction,
// unlink de l'ancien fichier APRÈS commit
let oldFilename = null;
db.transaction(() => {
  if (question_id) {
    const existing = db.prepare('SELECT filename FROM videos WHERE session_id=? AND question_id=?')
      .get(session_id, question_id);
    if (existing) {
      oldFilename = existing.filename;
      db.prepare('DELETE FROM videos WHERE session_id=? AND question_id=?').run(session_id, question_id);
    }
  }
  db.prepare(`INSERT INTO videos (id, session_id, question_id, question_text, filename, mime_type, size, checksum, recorded_at)
              VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)`).run(
    id, session_id, question_id ?? null, question_text, filename, mime_type, size, checksum,
    recorded_at ?? new Date().toISOString());
})();

// unlink APRÈS commit (§11.9)
if (oldFilename) {
  unlink(join(dataDir, 'events', active.id, 'videos', oldFilename), () => {});
}
⛔ Invariant §11.9 : DELETE+INSERT en transaction, unlink APRÈS le commit

Un invité peut ré-enregistrer sa réponse à une question (la contrainte UNIQUE(session_id, question_id) du schéma n'autorise qu'une vidéo par couple). Remplacer = supprimer l'ancienne ligne + insérer la nouvelle. Deux règles vitales :

  1. DELETE et INSERT dans une seule transaction. Sinon, entre les deux, la contrainte d'unicité pourrait être violée, ou un crash laisserait la session sans aucune vidéo pour cette question. La transaction rend le remplacement atomique : soit l'ancienne est remplacée par la nouvelle, soit rien ne change.
  2. unlink de l'ancien fichier APRÈS le commit. On mémorise oldFilename pendant la transaction, mais on ne supprime le fichier physique qu'après que la transaction a réussi. Pourquoi ? Si on supprimait le fichier avant ou pendant, et que la transaction échouait (rollback), on aurait perdu le fichier alors que la base pointe encore dessus → vidéo fantôme, données invité perdues. En supprimant après commit, le fichier n'est effacé que si la base a réellement basculé sur la nouvelle vidéo.

C'est l'ordre « base d'abord (atomique), fichier ensuite » — un invariant qu'on retrouve dans toute manipulation conjointe base+fichier du projet. Le casser introduit des incohérences base/disque très difficiles à diagnostiquer.

Le streaming Range-aware (§11.3)

Détail propre à la Borne : la route publique GET /sessions/:sessionId/videos/:questionId/file permet à l'invité de revoir sa propre vidéo (via le capability token de session), tandis que les routes /videos/:id/file et /download sont réservées à l'admin (avec auth). Le download ajoute Content-Disposition: attachment pour forcer le téléchargement.
