4 — Le serveur Borne

borne/routes/events.js

La gestion des événements côté Borne : création locale, activation, clôture, et deux routes particulières — preflight (vérifications avant ouverture du kiosque) et /event (l'événement actif servi au kiosque public). Le routeur mélange routes admin (protégées) et publiques.

Le pattern d'authentification injectée

export function makeEventsRouter(dataDir, cfg) {
  const router = Router();
  const auth = cfg.requireAdmin;            // middleware injecté (voir borne/index.js)

  router.get('/events', auth, (req, res) => { res.json(listEvents()); });   // admin
  router.get('/event', (req, res, next) => { /* … */ });                    // PUBLIC (pas de auth)
  // …

cfg.requireAdmin est le middleware d'auth « armé » passé par index.js. Les routes admin le placent en 2ᵉ argument (router.get('/events', auth, handler)) ; les routes publiques (kiosque) ne l'ont pas. C'est la distinction admin/public, lisible route par route.

Création d'un événement local

router.post('/events', auth, (req, res, next) => {
  try {
    const { name, event_date } = req.body;
    if (!name || typeof name !== 'string' || name.trim().length === 0) {
      return res.status(400).json({ error: 'Le nom de l\'événement est requis' });
    }
    const id = uuidv4();
    const eventDir = join(dataDir, 'events', id);
    mkdirSync(join(eventDir, 'videos'), { recursive: true });

    const db = createEventDb(join(eventDir, 'db.sqlite'));
    db.prepare(`INSERT OR IGNORE INTO event_meta (key, value) VALUES (?, ?)`).run('event_id', id);
    db.prepare(`INSERT OR IGNORE INTO event_meta (key, value) VALUES (?, ?)`).run('name', name.trim());
    db.prepare(`INSERT OR IGNORE INTO event_meta (key, value) VALUES (?, ?)`).run('origin', 'local');
    db.prepare(`INSERT OR IGNORE INTO event_meta (key, value) VALUES (?, ?)`).run('theme', DEFAULTS.THEME);
    if (event_date) db.prepare(`INSERT OR IGNORE INTO event_meta ...`).run('event_date', event_date);
    db.close();

    insertEvent({ id, name: name.trim(), origin: 'local' });
    const event = listEvents().find(e => e.id === id);
    res.status(201).json(event);
  } catch (err) { next(err); }
});

Un événement créé sur la Borne a l'origine 'local' (vs 'hub' pour un pullé). On crée le dossier + le db.sqlite (via createEventDb, qui seede les questions par défaut), on remplit event_meta (dont le thème par défaut DEFAULTS.THEME), puis on l'enregistre dans le registre local. INSERT OR IGNORE évite une erreur si la clé existe déjà (idempotence). Noter le db.close() explicite : ici on ouvre une base hors du cache eventDb (juste pour l'initialiser), donc on la ferme manuellement.

Régler le thème : PUT /events/:id/settings

// Réglages d'événement (config, pas donnée invité → conforme RGPD §11).
// Pour l'instant : le thème visuel du parcours invité. Écrit dans event_meta.
router.put('/events/:id/settings', auth, (req, res, next) => {
  try {
    const { id } = req.params;
    const { theme } = req.body;
    const event = listEvents().find(e => e.id === id);
    if (!event) return res.status(404).json({ error: 'Événement introuvable' });

    if (theme !== undefined && !THEMES.includes(theme)) {
      return res.status(400).json({ error: `Thème invalide (attendu : ${THEMES.join(', ')})` });
    }

    // Le thème ne concerne que l'événement vu par les invités (l'actif).
    // Restreindre à l'actif évite aussi que getActiveEventDb (cache à un slot)
    // ne ferme le handle de l'actif en ouvrant celui d'un autre événement.
    const activeEvent = getActiveEvent();
    if (!activeEvent || activeEvent.id !== id) {
      return res.status(409).json({ error: 'Seul l\'événement actif peut être configuré' });
    }

    const db = getActiveEventDb(dataDir, activeEvent);
    if (theme !== undefined) {
      db.prepare(`INSERT INTO event_meta (key, value) VALUES ('theme', ?)
                  ON CONFLICT(key) DO UPDATE SET value=excluded.value`).run(theme);
    }
    const getMeta = (key) => db.prepare('SELECT value FROM event_meta WHERE key=?').get(key)?.value ?? null;
    res.json({ id, theme: getMeta('theme') ?? DEFAULTS.THEME });
  } catch (err) { next(err); }
});

Cette route admin permet de changer le thème visuel du parcours invité (sélectionné depuis le panneau admin). Quatre gardes, dans l'ordre : 404 si l'id n'existe pas, 400 si le thème reçu n'est pas dans la liste blanche THEMES, 409 si l'événement n'est pas l'actif, 401 si pas authentifié (le middleware auth en amont). L'écriture se fait en upsert (ON CONFLICT … DO UPDATE) sur la clé theme de event_meta.

ℹ Pourquoi restreindre à l'événement actif (409) ?

Deux raisons. Fonctionnelle : seul l'événement actif est vu par les invités, donc seul son thème a un effet visible. Technique : le cache eventDb de la Borne n'a qu'un seul slot. Ouvrir le db.sqlite d'un autre événement via getActiveEventDb fermerait le handle de l'actif — d'où la restriction, qui garde le cache cohérent.

💡 Le thème est une config, pas une donnée invité

Le thème vit dans event_meta de events/<id>/db.sqlite, jamais dans registry.sqlite — comme toute config d'événement, et conformément à l'invariant RGPD §11.13. Côté kiosque il est appliqué via data-theme sur <html> (recalcul de variables CSS, sans re-render) ; le bouton REC reste rouge dans tous les thèmes.

Le preflight : tout vérifier avant d'ouvrir le kiosque

router.get('/preflight', auth, async (req, res, next) => {
  try {
    const activeEvent = getActiveEvent();
    const stats = await statfs(dataDir);
    const free_bytes = stats.bfree * stats.bsize;
    const disk_ok = free_bytes > LIMITS.DISK_ALERT_BYTES;     // assez d'espace ?

    let event_info = { loaded: false, pulled_at: null };
    let questions_count = 0;
    if (activeEvent) {
      event_info = { loaded: true, pulled_at: activeEvent.pulled_at ?? null };
      const db = getActiveEventDb(dataDir, activeEvent);
      questions_count = db.prepare('SELECT COUNT(*) as n FROM questions WHERE enabled=1').get().n;
    }

    let clock_ok = null;                                       // dérive d'horloge ?
    const client_time = req.query.client_time;
    if (client_time) {
      const clientMs = new Date(client_time).getTime();
      if (!isNaN(clientMs)) clock_ok = Math.abs(Date.now() - clientMs) < 2 * 60 * 1000;
    }
    res.json({ event: event_info, questions_count, disk_ok, clock_ok });
  } catch (err) { next(err); }
});
JS statfs — l'espace disque, et la comparaison d'horloge

statfs(dataDir) (de node:fs/promises, donc await) renvoie des stats du système de fichiers : bfree (blocs libres) × bsize (taille d'un bloc) = octets libres. On compare au seuil LIMITS.DISK_ALERT_BYTES (10 Go) : si la borne n'a pas assez d'espace pour enregistrer des vidéos, disk_ok est faux et le kiosque peut alerter avant l'événement.

Le clock_ok compare l'heure du client (l'iPad) à celle du serveur : si l'écart dépasse 2 minutes, l'horloge de l'un des deux dérive (problème pour les horodatages, surtout sur une borne hors-ligne sans NTP). Math.abs(Date.now() - clientMs) < 2*60*1000 : Date.now() est l'heure serveur en millisecondes ; isNaN garde contre une date client invalide.

ℹ Pourquoi un preflight ?

Une borne en événement est souvent hors-ligne et sans surveillance. Le preflight permet à l'opérateur de vérifier avant l'arrivée des invités que tout est prêt : un événement est chargé, il a des questions actives, le disque a de la place, l'horloge est juste. Mieux vaut détecter un problème à l'installation qu'en plein événement.

La route publique /event (kiosque)

router.get('/event', (req, res, next) => {            // PAS de auth — accès kiosque
  const activeEvent = getActiveEvent();
  if (!activeEvent) return res.status(404).json({ error: 'Aucun événement actif' });

  const db = getActiveEventDb(dataDir, activeEvent);
  const getMeta = (key) => db.prepare('SELECT value FROM event_meta WHERE key=?').get(key)?.value ?? null;

  res.json({
    id: activeEvent.id, name: activeEvent.name, status: activeEvent.status,
    consent_text: getMeta('consent_text') ?? DEFAULTS.CONSENT_TEXT,
    idle_timeout: parseInt(getMeta('idle_timeout') ?? String(DEFAULTS.IDLE_TIMEOUT_S), 10),
    theme: getMeta('theme') ?? DEFAULTS.THEME,
  });
});

C'est ce que le kiosque (l'iPad) interroge pour savoir quel événement afficher : nom, texte de consentement, délai d'inactivité, et le thème (appliqué au chargement en data-theme sur <html>). Les valeurs viennent de event_meta, avec un repli sur les DEFAULTS si non définies. getMeta est un petit helper local (closure sur db) qui lit une clé meta, avec optional chaining + nullish pour gérer l'absence.
